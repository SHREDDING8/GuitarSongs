const normArr = (v) => {
  if (v == null) return [];
  if (Array.isArray(v)) return v.filter((x) => x != null && String(x).trim() !== "");
  const s = String(v).trim();
  return s ? [s] : [];
};

const normStr = (v) => {
  if (v == null) return "";

const toText = (v) => String(v ?? "").trim();

const uniqSorted = (arr) =>

    a.localeCompare(b, "ru", { sensitivity: "base" })

      title,
      artist: normStr(p[F.artist]),
      tuning: normStr(p[F.tuning]),
      notesType: normArr(p[F.notesType]),
      playType: normArr(p[F.playType]),
      instruments: normArr(p[F.instruments]),
      attachments: normArr(p[F.attachments]),
      embeds: (p.file.embeds ?? []).map((l) => l?.path).filter(Boolean),
      created: p["Дата создания"] ?? p.file.ctime,
    };
  })
  .array();

const mkField = (label, inputEl, extraClass = "") => {
  const wrap = document.createElement("div");
  wrap.className = `gtr-field ${extraClass}`.trim();
  const lab = document.createElement("label");
  lab.textContent = label;
  wrap.appendChild(lab);
  wrap.appendChild(inputEl);
  return wrap;
};

const mkSelect = ({ multiple = false, placeholder = "Все", options = [] } = {}) => {
  const sel = document.createElement("select");
  if (multiple) sel.multiple = true;
  if (!multiple) {
    const opt = document.createElement("option");
    opt.value = "";
    opt.textContent = placeholder;
    sel.appendChild(opt);
  }
  for (const item of options) {
    const v = typeof item === "string" ? item : item.value;
    const label = typeof item === "string" ? item : item.label;
    const opt = document.createElement("option");
    opt.value = v;
    opt.textContent = label;
    sel.appendChild(opt);
  }
  return sel;
};

const mkInput = (placeholder) => {
  const inp = document.createElement("input");
  inp.type = "text";
  inp.placeholder = placeholder;
  return inp;
};

const getMulti = (sel) => [...sel.selectedOptions].map((o) => o.value).filter(Boolean);
const withEmptyOption = (values) => [{ value: EMPTY_VALUE, label: EMPTY_LABEL }, ...values];
const mkMultiDropdown = (sel, placeholder = "Выбери...") => {
  sel.classList.add("gtr-select-native");

  const wrap = document.createElement("div");
  wrap.className = "gtr-dd";

  const btn = document.createElement("button");
  btn.type = "button";
  btn.className = "gtr-dd-btn";
  const btnText = document.createElement("span");
  btnText.className = "gtr-dd-text gtr-dd-placeholder";
  btnText.textContent = placeholder;
  btn.appendChild(btnText);

  const menu = document.createElement("div");
  menu.className = "gtr-dd-menu";

  const updateButtonText = () => {
    const selected = getMulti(sel);
    if (!selected.length) {
      btnText.textContent = placeholder;
      btnText.classList.add("gtr-dd-placeholder");
      return;
    }
    const labels = [...sel.selectedOptions].map((o) => o.textContent);
    btnText.textContent = labels.join(", ");
    btnText.classList.remove("gtr-dd-placeholder");
  };

  const close = () => wrap.classList.remove("open");

  for (const opt of sel.options) {
    const row = document.createElement("label");
    row.className = "gtr-dd-opt";
    if (opt.selected) row.classList.add("is-selected");
    const cb = document.createElement("input");
    cb.type = "checkbox";
    cb.checked = opt.selected;
    const checkMark = document.createElement("span");
    checkMark.className = "gtr-dd-check";
    checkMark.textContent = "✓";
    const text = document.createElement("span");
    text.className = "gtr-dd-label";
    text.textContent = opt.textContent;
    row.appendChild(checkMark);
    row.appendChild(cb);
    row.appendChild(text);
    cb.addEventListener("change", () => {
      opt.selected = cb.checked;
      row.classList.toggle("is-selected", cb.checked);
      sel.dispatchEvent(new Event("change"));
    });
    menu.appendChild(row);
  }

  btn.addEventListener("click", (e) => {
    e.preventDefault();
    wrap.classList.toggle("open");
  });

  document.addEventListener("click", (e) => {
    if (!wrap.contains(e.target)) close();
  });

  sel.addEventListener("change", () => {
    const opts = [...sel.options];
    const rows = [...menu.querySelectorAll(".gtr-dd-opt")];
    const checks = [...menu.querySelectorAll('input[type="checkbox"]')];
    for (let i = 0; i < opts.length; i++) checks[i].checked = opts[i].selected;
    for (let i = 0; i < opts.length; i++) rows[i].classList.toggle("is-selected", opts[i].selected);
    updateButtonText();
  });

  wrap.appendChild(btn);
  wrap.appendChild(menu);
  wrap.appendChild(sel);
  updateButtonText();
  return wrap;
};

const matchesAny = (arr, wanted) => {
  if (!wanted.length) return true;
  const set = new Set(arr.map((x) => toText(x)));
  return wanted.some((w) => set.has(w));
};

const matchesText = (text, q) => {
  if (!q) return true;
  return toText(text).toLowerCase().includes(q.toLowerCase());
};
const matchesSelected = (values, wanted) => {
  if (!wanted.length) return true;
  const normalizedValues = normArr(values).map((v) => toText(v));
  const wantsEmpty = wanted.includes(EMPTY_VALUE);
  const concreteWanted = wanted.filter((w) => w !== EMPTY_VALUE);
  const hasConcreteMatch = concreteWanted.length
    ? concreteWanted.some((w) => normalizedValues.includes(w))
    : false;
  const isEmpty = normalizedValues.length === 0;
  if (wantsEmpty && concreteWanted.length) return isEmpty || hasConcreteMatch;
  if (wantsEmpty) return isEmpty;
  return hasConcreteMatch;
};

const render = () => {
  const q = qInput.value.trim();
  const artistWanted = getMulti(artistSel);
  const tuningWanted = getMulti(tuningSel);
  const notesWanted = getMulti(notesSel);
  const playWanted = getMulti(playSel);
  const instWanted = getMulti(instSel);
  const attWanted = getMulti(attSel);

  const filtered = pages
    .filter((p) => matchesText(p.title, q))
    .filter((p) => matchesSelected(p.artist, artistWanted))
    .filter((p) => matchesSelected(p.tuning, tuningWanted))
    .filter((p) => matchesSelected(p.notesType, notesWanted))
    .filter((p) => matchesSelected(p.playType, playWanted))
    .filter((p) => matchesSelected(p.instruments, instWanted))
    .filter((p) => matchesSelected(p.attachments, attWanted))
    .sort((a, b) => (a.artist || "").localeCompare(b.artist || "", "ru", { sensitivity: "base" }) || a.title.localeCompare(b.title, "ru", { sensitivity: "base" }));

  meta.textContent = `Найдено: ${filtered.length} из ${pages.length}`;

  grid.innerHTML = "";
  if (!filtered.length) {
    const empty = document.createElement("div");
    empty.className = "gtr-empty";
    empty.textContent = "Ничего не найдено. Попробуй очистить фильтры или изменить запрос.";
    grid.appendChild(empty);
    return;
  }

  for (const p of filtered) {
    const card = document.createElement("div");
    card.className = "gtr-card";

    const titleRow = document.createElement("div");
    titleRow.className = "gtr-title";
    const linkWrap = document.createElement("div");
    dv.el("span", dv.fileLink(p.page.file.path, false, p.title), { container: linkWrap });
    titleRow.appendChild(linkWrap);
    const right = document.createElement("div");
    right.style.color = "var(--muted)";
    right.style.fontSize = "12px";
    right.textContent = p.tuning ? p.tuning : "";
    titleRow.appendChild(right);
    card.appendChild(titleRow);

    const sub = document.createElement("div");
    sub.className = "gtr-sub";
    sub.textContent = p.artist ? p.artist : "—";
    card.appendChild(sub);

    const kv = document.createElement("div");
    kv.className = "gtr-kv";

    const addRow = (k, values) => {
      const vArr = normArr(values);
      const kEl = document.createElement("div");
      kEl.className = "gtr-k";
      kEl.textContent = k;
      const vEl = document.createElement("div");
      vEl.className = "gtr-v";
      if (vArr.length) {
        const pills = document.createElement("div");
        pills.className = "gtr-pills";
        for (const it of vArr) {
          const pill = document.createElement("span");
          pill.className = "gtr-pill";
          pill.textContent = toText(it);
          pills.appendChild(pill);
        }
        vEl.appendChild(pills);
      } else {
        vEl.textContent = "—";
        vEl.style.color = "var(--muted)";
        vEl.style.fontSize = "12px";
      }
      kv.appendChild(kEl);
      kv.appendChild(vEl);
    };

    addRow("Тип нот", p.notesType);
    addRow("Тип игры", p.playType);
    addRow("Инструменты", p.instruments);
    addRow("Вложения (тип)", p.attachments);
    card.appendChild(kv);

    // Файлы-вложения (фактические embeds из тела заметки)
    const embeds = uniqSorted(p.embeds);
    if (embeds.length) {
      const files = document.createElement("div");
      files.className = "gtr-files";
      for (const path of embeds) {
        const fn = path.split("/").pop();
        const chip = document.createElement("span");
        chip.className = "gtr-file";
        dv.el("span", dv.fileLink(path, false, fn), { container: chip });
        files.appendChild(chip);
      }
      card.appendChild(files);
    }

    grid.appendChild(card);
  }
};

const onChange = () => render();
for (const el of [qInput, artistSel, tuningSel, notesSel, playSel, instSel, attSel]) {

    for (const opt of s.options) opt.selected = false;
    s.dispatchEvent(new Event("change"));
  }
  render();
});
